import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Your tests, checked">
    Coverage says a line ran. mutastic changes that line, `>=` to `>` say, and runs the tests that
    reach it. If they all still pass, nothing checks what the line does.
  </Card>
  <Card title="The test to write">
    Each missed mutant comes with the inputs your tests used, the gap they left, a test that would
    close it, and the command that proves it closed.
  </Card>
  <Card title="Three ways to run it">
    The `mutastic` command, `mvn mutastic:run`, or `./gradlew mutasticRun`. The same run, the same
    results, the same options.
  </Card>
  <Card title="Made for agents too">
    `--format=agent` gives coding agents the verdict first, then each gap and the rule that a fix
    changes tests, never production code.
  </Card>
</CardGrid>
